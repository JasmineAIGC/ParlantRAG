

## 设计目标

```
设计约束：
  1. 对外只暴露一个注册函数：register_rag_module(agent)
  2. 检索器（Retriever）可插拔替换
  3. 评价器（Evaluator）可插拔替换
  4. 内部实现对宿主系统完全透明
  5. 不侵入宿主系统的任何现有逻辑

使用方式（目标）：
  # 宿主系统只需两行
  rag = RAGModule(retriever=MyRetriever(), evaluator=MyEvaluator())
  await rag.register(agent)
```

### base.py — 抽象接口

```python name=rag_module/base.py
"""
抽象接口层。
所有的检索器和评价器都必须实现这两个接口。
宿主系统只需要实现接口，不需要了解模块内部细节。
"""
from abc import ABC, abstractmethod
from dataclasses import dataclass, field


@dataclass
class Chunk:
    """一个检索到的文档片段"""
    chunk_id: str
    content: str
    source: str
    score: float = 0.0
    metadata: dict = field(default_factory=dict)


@dataclass
class RetrievalResult:
    """检索结果"""
    query: str
    chunks: list[Chunk]
    status: str  # "success" | "no_results"


@dataclass
class EvaluationResult:
    """评价结果"""
    verdict: str          # "qualified" | "insufficient" | "no_coverage"
    relevance: float      # 相关性得分 0~1
    coverage: float       # 覆盖度得分 0~1
    confidence: float     # 综合置信度 0~1
    conclusion: str       # 人类可读的结论
    action_hint: str      # 给 Agent 的后续行动提示
    details: dict = field(default_factory=dict)


class BaseRetriever(ABC):
    """
    检索器接口。
    实现此接口即可接入任意向量数据库或检索系统。
    """

    @abstractmethod
    async def retrieve(self, query: str, top_k: int = 3) -> RetrievalResult:
        """
        从知识库检索与 query 最相关的片段。

        Args:
            query:  检索查询词
            top_k:  返回片段数量上限

        Returns:
            RetrievalResult
        """
        ...


class BaseEvaluator(ABC):
    """
    评价器接口。
    实现此接口即可替换评价逻辑（规则评价 / LLM 评价 / 混合评价）。
    """

    @abstractmethod
    async def evaluate(
        self,
        question: str,
        retrieval: RetrievalResult,
    ) -> EvaluationResult:
        """
        评价检索结果的质量。

        Args:
            question:  用户的原始问题
            retrieval: 检索器返回的结果

        Returns:
            EvaluationResult
        """
        ...
```

### retriever.py — 内置检索器

```python name=rag_module/retriever.py
"""
内置检索器实现。
提供两个开箱即用的实现：
  - KeywordRetriever：基于关键词匹配，无需额外依赖，适合快速验证
  - DenseRetriever：基于向量检索，需要 sentence-transformers，适合生产

宿主系统也可以完全忽略这个文件，自己实现 BaseRetriever。
"""
from .base import BaseRetriever, Chunk, RetrievalResult


class KeywordRetriever(BaseRetriever):
    """
    关键词检索器（开箱即用，无需额外依赖）。
    适合：快速原型验证、小规模知识库、不想引入向量库的场景。

    Usage:
        docs = [{"id": "1", "content": "...", "source": "file.md"}]
        retriever = KeywordRetriever(docs)
    """

    def __init__(self, documents: list[dict]) -> None:
        """
        Args:
            documents: 文档列表，每条必须包含 id / content / source 字段
        """
        self._docs = documents

    async def retrieve(self, query: str, top_k: int = 3) -> RetrievalResult:
        query_words = set(query.lower().split())
        scored = []

        for doc in self._docs:
            doc_words = set(doc["content"].lower().split())
            union = len(query_words | doc_words)
            intersection = len(query_words & doc_words)
            score = round(intersection / union, 4) if union > 0 else 0.0
            scored.append((doc, score))

        scored.sort(key=lambda x: x[1], reverse=True)
        top = scored[:top_k]

        chunks = [
            Chunk(
                chunk_id=doc["id"],
                content=doc["content"],
                source=doc["source"],
                score=score,
                metadata=doc.get("metadata", {}),
            )
            for doc, score in top
            if score > 0
        ]

        return RetrievalResult(
            query=query,
            chunks=chunks,
            status="success" if chunks else "no_results",
        )


class DenseRetriever(BaseRetriever):
    """
    向量检索器（需要 sentence-transformers + faiss）。
    适合：生产环境、大规模知识库、需要语义匹配的场景。

    Usage:
        retriever = DenseRetriever(
            documents=docs,
            model_name="paraphrase-multilingual-MiniLM-L12-v2",
        )
        await retriever.build_index()  # 启动时调用一次
    """

    def __init__(
        self,
        documents: list[dict],
        model_name: str = "paraphrase-multilingual-MiniLM-L12-v2",
    ) -> None:
        self._docs = documents
        self._model_name = model_name
        self._index = None
        self._model = None

    async def build_index(self) -> None:
        """构建向量索引，启动时调用一次"""
        try:
            from sentence_transformers import SentenceTransformer
            import faiss
            import numpy as np
        except ImportError:
            raise ImportError(
                "DenseRetriever 需要安装额外依赖：\n"
                "pip install sentence-transformers faiss-cpu"
            )

        self._model = SentenceTransformer(self._model_name)
        texts = [doc["content"] for doc in self._docs]
        embeddings = self._model.encode(texts, normalize_embeddings=True)

        dim = embeddings.shape[1]
        self._index = faiss.IndexFlatIP(dim)
        self._index.add(embeddings.astype(np.float32))

    async def retrieve(self, query: str, top_k: int = 3) -> RetrievalResult:
        if self._index is None or self._model is None:
            raise RuntimeError("请先调用 await retriever.build_index()")

        import numpy as np

        query_vec = self._model.encode(
            [query], normalize_embeddings=True
        ).astype(np.float32)

        scores, indices = self._index.search(query_vec, top_k)

        chunks = [
            Chunk(
                chunk_id=self._docs[idx]["id"],
                content=self._docs[idx]["content"],
                source=self._docs[idx]["source"],
                score=round(float(scores[0][i]), 4),
                metadata=self._docs[idx].get("metadata", {}),
            )
            for i, idx in enumerate(indices[0])
            if idx >= 0 and scores[0][i] > 0
        ]

        return RetrievalResult(
            query=query,
            chunks=chunks,
            status="success" if chunks else "no_results",
        )
```

### evaluator.py — 内置评价器

```python name=rag_module/evaluator.py
"""
内置评价器实现。
提供两个开箱即用的实现：
  - RuleBasedEvaluator：基于关键词统计规则，无需 LLM，速度快、成本低
  - LLMEvaluator：调用 LLM 进行语义级别评价，准确度更高

宿主系统也可以完全忽略这个文件，自己实现 BaseEvaluator。
"""
from .base import BaseEvaluator, EvaluationResult, RetrievalResult

# 评价结论阈值（可通过子类覆盖调整）
QUALIFIED_THRESHOLD = 0.45
INSUFFICIENT_THRESHOLD = 0.20

STOPWORDS = {
    "的", "了", "是", "在", "我", "有", "和", "与", "或", "吗", "呢", "吧",
    "what", "how", "why", "is", "the", "a", "an", "of", "in", "to", "do",
}


class RuleBasedEvaluator(BaseEvaluator):
    """
    规则评价器（开箱即用，无需 LLM）。
    基于关键词覆盖率和相关度分数计算三个维度的得分。

    适合：对速度和成本敏感、评价精度要求不极高的场景。
    """

    def __init__(
        self,
        qualified_threshold: float = QUALIFIED_THRESHOLD,
        insufficient_threshold: float = INSUFFICIENT_THRESHOLD,
    ) -> None:
        self._qualified_threshold = qualified_threshold
        self._insufficient_threshold = insufficient_threshold

    async def evaluate(
        self,
        question: str,
        retrieval: RetrievalResult,
    ) -> EvaluationResult:

        # ── 无结果直接返回 ─────────────────────────────────
        if not retrieval.chunks:
            return EvaluationResult(
                verdict="no_coverage",
                relevance=0.0,
                coverage=0.0,
                confidence=0.0,
                conclusion="知识库中未找到与该问题相关的任何内容。",
                action_hint="告知用户该问题超出当前知识库的覆盖范围。",
            )

        question_words = set(question.lower().split()) - STOPWORDS

        # ── 维度一：相关性 ──────────────────────────────────
        # 每个片段与问题的平均匹配程度
        chunk_relevances = []
        for chunk in retrieval.chunks:
            chunk_words = set(chunk.content.lower().split())
            overlap = len(question_words & chunk_words)
            keyword_ratio = overlap / len(question_words) if question_words else 0.0
            # 结合检索时的向量得分（若有）
            combined = round(keyword_ratio * 0.5 + chunk.score * 0.5, 3)
            chunk_relevances.append(combined)

        avg_relevance = round(sum(chunk_relevances) / len(chunk_relevances), 3)

        # ── 维度二：覆盖度 ──────────────────────────────────
        # 所有片段合并后，覆盖了问题中多少关键词
        all_content = " ".join(c.content.lower() for c in retrieval.chunks)
        all_words = set(all_content.split())
        covered = question_words & all_words
        coverage = round(len(covered) / len(question_words), 3) if question_words else 0.0
        coverage_gaps = list(question_words - all_words)

        # ── 维度三：置信度（加权综合） ─────────────────────
        valid_count = sum(1 for r in chunk_relevances if r >= 0.15)
        valid_ratio = valid_count / len(retrieval.chunks)
        confidence = round(
            avg_relevance * 0.45 + coverage * 0.35 + valid_ratio * 0.20, 3
        )

        # ── 评价结论 ────────────────────────────────────────
        if confidence >= self._qualified_threshold:
            verdict = "qualified"
            conclusion = (
                f"检索结果质量合格。{len(retrieval.chunks)} 个片段中 "
                f"{valid_count} 个相关，关键词覆盖率 {coverage:.0%}。"
            )
            action_hint = "可直接基于检索内容作答，并注明信息来源。"

        elif confidence >= self._insufficient_threshold:
            verdict = "insufficient"
            conclusion = (
                f"检索结果相关性不足（置信度 {confidence:.2f}），"
                f"关键词覆盖率仅 {coverage:.0%}。"
                + (f"未覆盖关键词：{', '.join(coverage_gaps[:5])}" if coverage_gaps else "")
            )
            action_hint = "建议用更具体的关键词重新提问，或告知用户信息覆盖有限。"

        else:
            verdict = "no_coverage"
            conclusion = (
                f"知识库对该问题几乎没有覆盖（置信度 {confidence:.2f}），"
                f"无法给出可靠回答。"
            )
            action_hint = "告知用户该问题超出知识库范围，建议联系人工支持。"

        return EvaluationResult(
            verdict=verdict,
            relevance=avg_relevance,
            coverage=coverage,
            confidence=confidence,
            conclusion=conclusion,
            action_hint=action_hint,
            details={
                "chunk_relevances": chunk_relevances,
                "coverage_gaps": coverage_gaps,
                "valid_chunk_count": valid_count,
            },
        )


class LLMEvaluator(BaseEvaluator):
    """
    LLM 评价器（需要 OpenAI API）。
    使用 LLM 进行语义级别的相关性判断，准确度更高，但有额外成本。

    适合：对评价质量要求高、允许额外 LLM 调用成本的场景。

    Usage:
        evaluator = LLMEvaluator(api_key="sk-...", model="gpt-4o-mini")
    """

    def __init__(
        self,
        api_key: str,
        model: str = "gpt-4o-mini",
        base_url: str | None = None,
    ) -> None:
        self._api_key = api_key
        self._model = model
        self._base_url = base_url

    async def evaluate(
        self,
        question: str,
        retrieval: RetrievalResult,
    ) -> EvaluationResult:
        try:
            from openai import AsyncOpenAI
        except ImportError:
            raise ImportError("LLMEvaluator 需要安装：pip install openai")

        if not retrieval.chunks:
            return EvaluationResult(
                verdict="no_coverage",
                relevance=0.0,
                coverage=0.0,
                confidence=0.0,
                conclusion="未检索到任何内容。",
                action_hint="告知用户该问题超出知识库范围。",
            )

        client = AsyncOpenAI(api_key=self._api_key, base_url=self._base_url)
        chunks_text = "\n\n".join(
            f"[片段{i+1}] 来源：{c.source}\n{c.content}"
            for i, c in enumerate(retrieval.chunks)
        )

        prompt = f"""你是一个 RAG 检索质量评估专家。

用户问题：{question}

检索到的内容片段：
{chunks_text}

请评估这批检索内容的质量，返回严格的 JSON 格式：
{{
  "relevance": 0到1的小数（片段与问题的平均相关程度）,
  "coverage": 0到1的小数（片段能覆盖问题多少方面）,
  "confidence": 0到1的小数（基于这批内容能否可靠作答）,
  "verdict": "qualified"（可靠作答）或 "insufficient"（信息不足）或 "no_coverage"（无覆盖）,
  "conclusion": "一句话评价结论",
  "action_hint": "给AI助手的建议"
}}

只返回 JSON，不要其他文字。"""

        response = await client.chat.completions.create(
            model=self._model,
            messages=[{"role": "user", "content": prompt}],
            response_format={"type": "json_object"},
        )

        import json
        result = json.loads(response.choices[0].message.content)

        return EvaluationResult(
            verdict=result.get("verdict", "no_coverage"),
            relevance=float(result.get("relevance", 0.0)),
            coverage=float(result.get("coverage", 0.0)),
            confidence=float(result.get("confidence", 0.0)),
            conclusion=result.get("conclusion", ""),
            action_hint=result.get("action_hint", ""),
        )
```

### tools.py — Parlant 工具定义

```python name=rag_module/tools.py
"""
Parlant 工具层。
将 Retriever 和 Evaluator 包装为 Parlant 工具。
这一层是 RAG 模块与 Parlant 框架之间的唯一接口。
"""
import json
from typing import Annotated

import parlant.sdk as p

from .base import BaseRetriever, BaseEvaluator


def make_rag_tools(
    retriever: BaseRetriever,
    evaluator: BaseEvaluator,
) -> tuple[p.ToolEntry, p.ToolEntry]:
    """
    根据传入的检索器和评价器，动态生成两个 Parlant 工具。

    Returns:
        (rag_search_tool, evaluate_retrieval_tool)
    """

    @p.tool
    async def rag_search(
        context: p.ToolContext,
        query: Annotated[
            str,
            p.ToolParameterOptions(
                description="从用户问题中提炼的核心检索词"
            ),
        ],
        top_k: Annotated[
            int,
            p.ToolParameterOptions(
                description="返回片段数量上限，默认 3"
            ),
        ] = 3,
    ) -> p.ToolResult:
        """从知识库检索与查询最相关的文档片段"""
        result = await retriever.retrieve(query=query, top_k=top_k)

        return p.ToolResult(
            data={
                "query": result.query,
                "retrieval_status": result.status,
                "chunk_count": len(result.chunks),
                "retrieved_chunks": [
                    {
                        "chunk_id": c.chunk_id,
                        "content": c.content,
                        "source": c.source,
                        "score": c.score,
                    }
                    for c in result.chunks
                ],
            }
        )

    @p.tool
    async def evaluate_retrieval(
        context: p.ToolContext,
        user_question: Annotated[
            str,
            p.ToolParameterOptions(description="用户的原始问题"),
        ],
        retrieved_chunks_json: Annotated[
            str,
            p.ToolParameterOptions(
                description="rag_search 返回的 retrieved_chunks 字段，序列化为 JSON 字符串"
            ),
        ],
    ) -> p.ToolResult:
        """评价 RAG 检索结果的质量（相关性、覆盖度、置信度）"""
        from .base import Chunk, RetrievalResult

        try:
            raw_chunks = json.loads(retrieved_chunks_json)
        except (json.JSONDecodeError, TypeError):
            raw_chunks = []

        # 将原始字典还原为 Chunk 对象，交给评价器
        chunks = [
            Chunk(
                chunk_id=c.get("chunk_id", ""),
                content=c.get("content", ""),
                source=c.get("source", ""),
                score=c.get("score", 0.0),
            )
            for c in raw_chunks
        ]
        retrieval = RetrievalResult(
            query=user_question,
            chunks=chunks,
            status="success" if chunks else "no_results",
        )

        eval_result = await evaluator.evaluate(
            question=user_question,
            retrieval=retrieval,
        )

        return p.ToolResult(
            data={
                "verdict": eval_result.verdict,
                "scores": {
                    "relevance": eval_result.relevance,
                    "coverage": eval_result.coverage,
                    "confidence": eval_result.confidence,
                },
                "conclusion": eval_result.conclusion,
                "action_hint": eval_result.action_hint,
                "details": eval_result.details,
            }
        )

    return rag_search, evaluate_retrieval
```

### module.py — RAGModule 主类

```python name=rag_module/module.py
"""
RAGModule 主类，对外唯一入口。
宿主系统只需要实例化这个类，然后调用 register(agent)。
"""
import parlant.sdk as p

from .base import BaseRetriever, BaseEvaluator
from .retriever import KeywordRetriever
from .evaluator import RuleBasedEvaluator
from .tools import make_rag_tools


class RAGModule:
    """
    RAG + 检索结果评价模块。

    设计原则：
      - 对宿主系统零侵入，只通过 register() 方法注入
      - 检索器和评价器完全可替换
      - Guidelines 的行为逻辑封装在模块内部

    Basic usage:
        rag = RAGModule(documents=my_docs)
        await rag.register(agent)

    Custom retriever/evaluator:
        rag = RAGModule(
            retriever=MyRetriever(),
            evaluator=MyEvaluator(),
        )
        await rag.register(agent)
    """

    def __init__(
        self,
        retriever: BaseRetriever | None = None,
        evaluator: BaseEvaluator | None = None,
        documents: list[dict] | None = None,
    ) -> None:
        """
        Args:
            retriever:  检索器实例，不传则使用内置 KeywordRetriever
            evaluator:  评价器实例，不传则使用内置 RuleBasedEvaluator
            documents:  当使用内置 KeywordRetriever 时，传入文档列表
                        格式：[{"id": "...", "content": "...", "source": "..."}]
        """
        if retriever is None:
            if documents is None:
                raise ValueError("未传入 retriever 时，必须提供 documents 参数")
            self._retriever = KeywordRetriever(documents)
        else:
            self._retriever = retriever

        self._evaluator = evaluator or RuleBasedEvaluator()

    async def register(self, agent: p.Agent) -> None:
        """
        将 RAG 工具和 Guidelines 注入到指定的 Parlant Agent。

        此方法是模块对外的唯一接口，调用后模块生效。
        可以在任意 Agent 上调用，不影响 Agent 的其他配置。

        Args:
            agent: 已创建的 Parlant Agent 实例
        """
        rag_search, evaluate_retrieval = make_rag_tools(
            retriever=self._retriever,
            evaluator=self._evaluator,
        )

        # Guideline 1：触发检索
        await agent.create_guideline(
            condition="用户提出了需要查询知识库才能回答的问题",
            action="""
            使用 rag_search 工具检索相关内容。
            从用户问题中提炼核心关键词作为 query。
            """,
            tools=[rag_search],
        )

        # Guideline 2：检索后评价
        await agent.create_guideline(
            condition="rag_search 工具已返回检索结果",
            action="""
            立即调用 evaluate_retrieval 工具评价本次检索结果。
            - user_question：用户的原始问题
            - retrieved_chunks_json：rag_search 返回的 retrieved_chunks 字段序列化为 JSON 字符串
            在拿到评价结论之前，不要开始作答。
            """,
            tools=[evaluate_retrieval],
        )

        # Guideline 3：评价合格 → 正常作答
        await agent.create_guideline(
            condition="evaluate_retrieval 返回的 verdict 为 qualified",
            action="""
            仅基于 rag_search 返回的检索内容作答，不添加检索结果之外的信息。
            在回答末尾注明信息来源文件名。
            """,
        )

        # Guideline 4：评价不足 → 告知用户
        await agent.create_guideline(
            condition="evaluate_retrieval 返回的 verdict 为 insufficient",
            action="""
            告知用户检索内容与问题的相关性不足，无法给出可靠回答。
            引用 evaluate_retrieval 的 conclusion 说明原因。
            建议用户用更具体的关键词重新提问。
            """,
        )

        # Guideline 5：无覆盖 → 诚实告知
        await agent.create_guideline(
            condition="""
            evaluate_retrieval 返回的 verdict 为 no_coverage，
            或 rag_search 返回的 retrieval_status 为 no_results
            """,
            action="""
            诚实告知用户：该问题超出了当前知识库的覆盖范围。
            不得凭自身知识作答。
            建议用户联系人工支持。
            """,
        )
```

### examples/custom\_usage.py — 自定义实现

```python name=rag_module/examples/custom_usage.py
"""
自定义示例：替换检索器和评价器。
"""
import asyncio
import parlant.sdk as p
from rag_module import RAGModule, BaseRetriever, BaseEvaluator
from rag_module import Chunk, RetrievalResult, EvaluationResult


# ── 自定义检索器（接入 Chroma / Pinecone / Weaviate 等）────────

class ChromaRetriever(BaseRetriever):
    def __init__(self, collection_name: str) -> None:
        import chromadb
        self._client = chromadb.Client()
        self._collection = self._client.get_collection(collection_name)

    async def retrieve(self, query: str, top_k: int = 3) -> RetrievalResult:
        results = self._collection.query(query_texts=[query], n_results=top_k)

        chunks = [
            Chunk(
                chunk_id=results["ids"][0][i],
                content=results["documents"][0][i],
                source=results["metadatas"][0][i].get("source", ""),
                score=1 - results["distances"][0][i],  # 距离转相似度
            )
            for i in range(len(results["ids"][0]))
        ]

        return RetrievalResult(
            query=query,
            chunks=chunks,
            status="success" if chunks else "no_results",
        )


# ── 自定义评价器（完全自定义评分逻辑）──────────────────────────

class StrictEvaluator(BaseEvaluator):
    """更严格的评价器：只有置信度超过 0.7 才算合格"""

    async def evaluate(self, question: str, retrieval: RetrievalResult) -> EvaluationResult:
        if not retrieval.chunks:
            return EvaluationResult(
                verdict="no_coverage",
                relevance=0.0, coverage=0.0, confidence=0.0,
                conclusion="未检索到内容。",
                action_hint="告知用户超出知识库范围。",
            )

        # 只看最高分片段的得分
        best_score = max(c.score for c in retrieval.chunks)
        avg_score = sum(c.score for c in retrieval.chunks) / len(retrieval.chunks)

        if best_score >= 0.7:
            return EvaluationResult(
                verdict="qualified",
                relevance=avg_score, coverage=best_score, confidence=best_score,
                conclusion=f"找到高质量匹配片段（得分 {best_score:.2f}）。",
                action_hint="可直接基于检索内容作答。",
            )
        else:
            return EvaluationResult(
                verdict="insufficient",
                relevance=avg_score, coverage=best_score, confidence=best_score,
                conclusion=f"最佳匹配得分仅 {best_score:.2f}，低于严格阈值 0.7。",
                action_hint="建议换词重新提问。",
            )


# ── 接入方式与内置实现完全一致 ──────────────────────────────────

async def main() -> None:
    async with p.Server() as server:
        agent = await server.create_agent(
            name="严格知识助手",
            description="只在高置信度时才作答的知识库助手。",
        )

        rag = RAGModule(
            retriever=ChromaRetriever("my_collection"),
            evaluator=StrictEvaluator(),
        )
        await rag.register(agent)

if __name__ == "__main__":
    asyncio.run(main())
```

## 各文件职责总结

```
文件                  职责                          可替换性
────────────────────────────────────────────────────────────
base.py               定义接口契约和数据结构          不变
retriever.py          内置检索器（可选使用）          可完全替换
evaluator.py          内置评价器（可选使用）          可完全替换
tools.py              将接口包装为 Parlant 工具       不变
module.py             组装 + 注册入口                不变
__init__.py           对外导出控制                   不变
```

```
宿主系统需要做的事：
  最简模式（3行）   → 传入 documents，调用 register(agent)
  自定义检索器      → 继承 BaseRetriever，实现 retrieve()
  自定义评价器      → 继承 BaseEvaluator，实现 evaluate()
  两者都自定义      → 各自继承，传入 RAGModule() 构造函数
```
